// propose関数内の計算ロジックを以下のように修正・強化しました function propose() { const list = document.getElementById('suggestion-list'); if (!state.crane || !state.radius || !state.weight) { list.innerHTML = `
条件を入力してください。
`; return; } const crane = CRANE_DB[state.crane]; const r_in = parseFloat(state.radius); const w_in = parseFloat(state.weight); // 必要揚程の計算 (着地高 + 荷物高 + 吊具長 + フック長) const h_max = parseFloat(state.h_max || 0); const h_load = parseFloat(state.h_load || 0); const h_tool = parseFloat(state.h_tool || 0); const requiredH = h_max + h_load + h_tool + crane.hookL.main; let results = []; // メインブーム計算 Object.keys(crane.outriggers).forEach(outKey => { const outSet = crane.outriggers[outKey]; crane.booms.forEach(len => { // --- 揚程チェック --- // 簡易三平方の定理で、そのブーム長で必要揚程に届くか判定 // (len^2 = radius^2 + height^2) const reachableH = Math.sqrt(Math.pow(len, 2) - Math.pow(r_in, 2)); if (isNaN(reachableH) || reachableH < requiredH) return; // 届かないブームは除外 const chart = outSet.data[len.toString()]; if(!chart) return; const dists = Object.keys(chart).map(Number).sort((a,b)=>a-b); let refR = null; for(let d of dists){ if(r_in <= d){ refR = d; break; } } if (refR !== null) { let rated = chart[refR.toFixed(1)] || chart[refR.toString()]; if(rated){ let totalW = w_in + crane.hookW.main; let ratio = (totalW / rated) * 100; let angle = Math.acos(r_in / len) * (180 / Math.PI); if (ratio < 110) { results.push({ isJib: false, val: len, ratio: Math.round(ratio), rated: rated, total: totalW, refR: refR, page: outSet.page, angle: angle.toFixed(1), parts: crane.boomParts[len.toString()], outName: outSet.name }); } } } }); }); // ジブ計算 (ジブは24mブーム固定の揚程で計算) // ※ジブの揚程計算はブーム角度とジブ長から算出 // ...（中略）... }
